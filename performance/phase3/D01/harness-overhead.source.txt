module d01_harness;
import benchmark.measurement : consume;
import npegged.storage : Status;
import core.stdc.stdio : printf;
import core.sys.posix.time : clock_gettime, CLOCK_MONOTONIC, timespec;

private ulong now() @nogc nothrow {
    timespec ts;
    return clock_gettime(CLOCK_MONOTONIC, &ts) ? 0 : cast(ulong)ts.tv_sec * 1_000_000_000 + ts
        .tv_nsec;
}

int main() @nogc nothrow {
    printf("sample,iterations,ns\n");
    foreach (iterations; [16, 256])
        foreach (sample; 0 .. 8) {
            const start = now();
            foreach (_; 0 .. iterations)
                consume(true, 0, 0, Status.ok);
            const elapsed = now() - start;
            if (!start || !elapsed)
                return 2;
            if (sample)
                printf("%u,%u,%llu\n", cast(uint)(sample - 1), cast(uint)iterations, elapsed);
        }
    return 0;
}
